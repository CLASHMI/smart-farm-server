-- 1. 创建数据库
CREATE DATABASE IF NOT EXISTS `smart_farm` DEFAULT CHARACTER SET utf8mb4;
USE `smart_farm`;

DROP TABLE IF EXISTS `sys_role`;
CREATE TABLE `sys_role` (
                            `id` BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键',
                            `role_name` VARCHAR(50) NOT NULL UNIQUE COMMENT '角色名称',
                            `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
                            `update_time` DATETIME DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
) COMMENT='角色表';

DROP TABLE IF EXISTS `sys_user`;
CREATE TABLE `sys_user` (
                            `id` BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键，自增',
                            `username` VARCHAR(20) NOT NULL UNIQUE COMMENT '登录账号 (YZ+序号)，唯一',
                            `password` VARCHAR(100) NOT NULL COMMENT '登录密码 (加密存储)',
                            `real_name` VARCHAR(10) NOT NULL COMMENT '人员姓名',
                            `role_id` BIGINT NOT NULL COMMENT '角色ID (关联sys_role.id)', -- 逻辑外键
                            `phone` VARCHAR(11) NOT NULL UNIQUE COMMENT '手机号 (唯一)',
                            `status` TINYINT(1) NOT NULL DEFAULT 0 COMMENT '状态 (0:正常, 1:停用)',
                            `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
                            `update_time` DATETIME DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
) COMMENT='人员/用户表';

DROP TABLE IF EXISTS `sys_dict_data`;
CREATE TABLE `sys_dict_data` (
                                 `id` BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键',
                                 `type_id` BIGINT NOT NULL COMMENT '关联的字典类型ID (关联 sys_dict_type.id)', -- 逻辑外键
                                 `dict_name` VARCHAR(100) NOT NULL COMMENT '字典选项名称', -- 替代 label 和 value 的统一名称字段
                                 `dict_sort` INT DEFAULT 0 COMMENT '排序号',
                                 `status` TINYINT(1) NOT NULL DEFAULT 0 COMMENT '状态 (0:正常, 1:停用)',
                                 `is_referenced` TINYINT(1) NOT NULL DEFAULT 0 COMMENT '是否被引用 (0:否, 1:是)',
                                 `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
                                 `update_time` DATETIME DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
                                 INDEX `idx_type_id` (`type_id`) -- 保留索引用于查询优化
) COMMENT='数据字典数据表';
